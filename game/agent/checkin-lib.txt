// checkin-lib.js — one-call check-in. Load: eval(bb.read('agent/checkin-lib.txt')); const r = await __checkin();
// Keeps autopilot + telemetry alive, scores predictions, computes next ETA, writes the snapshot, and returns a compact
// summary. `attention` lists the only things the agent must handle; if it is empty, just log + reschedule r.nextMin.
(() => {
  const readJ = (f) => { try { return JSON.parse(bb.read(f) || "null"); } catch { return null; } };
  const idb = (name, up) => new Promise((res, rej) => { const q = up ? indexedDB.open(name, 1) : indexedDB.open(name); if (up) q.onupgradeneeded = () => up(q.result); q.onsuccess = () => res(q.result); q.onerror = () => rej(q.error); });
  const reqP = (r) => new Promise((res, rej) => { r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
  // Restore (manual, user-approved only): copy agentBackups.saves[key].data into bitburnerSave.savestring["save"], then reload.
  window.__backupSave = async ({ everyMin = 60, force = false } = {}) => {
    const bk = await idb("agentBackups", (db) => db.createObjectStore("saves"));
    const keys = (await reqP(bk.transaction("saves").objectStore("saves").getAllKeys())).sort();
    const lastT = keys.length ? Date.parse(keys[keys.length - 1]) : 0;
    if (!force && Date.now() - lastT < everyMin * 60000) { bk.close(); return { saved: false, kept: keys.length }; }
    const src = await idb("bitburnerSave");
    const data = await reqP(src.transaction("savestring").objectStore("savestring").get("save")); src.close();
    if (!data) throw new Error("no save found");
    const key = new Date().toISOString(), store = bk.transaction("saves", "readwrite").objectStore("saves");
    await reqP(store.put({ t: Date.now(), data }, key));
    // retention: everything < 24h, plus the first backup of each day for 14 days
    const all = [...keys, key], now = Date.now(), days = new Set(), drop = [];
    for (const k of all) { const t = Date.parse(k), day = k.slice(0, 10);
      if (now - t < 864e5) continue; if (now - t < 14 * 864e5 && !days.has(day)) { days.add(day); continue; } drop.push(k); }
    const st2 = bk.transaction("saves", "readwrite").objectStore("saves"); for (const k of drop) st2.delete(k);
    bk.close();
    return { saved: true, key, bytes: data.length || data.byteLength || 0, kept: all.length - drop.length };
  };
  // Off-browser export every 6h into a user-picked folder (File System Access API). The save bytes are the same
  // gzip JSON as Options > Export, so files import back via Options > Import.
  const meta = () => idb("agentMeta", (db) => db.createObjectStore("kv"));
  const kvGet = async (k) => { const d = await meta(); const v = await reqP(d.transaction("kv").objectStore("kv").get(k)); d.close(); return v; };
  const kvSet = async (k, v) => { const d = await meta(); await reqP(d.transaction("kv", "readwrite").objectStore("kv").put(v, k)); d.close(); };
  window.__exportButton = () => {
    if (document.getElementById("agent-export-btn")) return "exists";
    const b = document.createElement("button");
    b.id = "agent-export-btn"; b.textContent = "Agent: pick backup folder";
    Object.assign(b.style, { position: "fixed", right: "12px", bottom: "12px", zIndex: 99999, padding: "10px 14px", background: "#0c0", color: "#000", font: "bold 14px monospace", border: "0", borderRadius: "6px", cursor: "pointer" });
    b.onclick = async () => {
      try {
        let h = await kvGet("dir");
        if (h && (await h.requestPermission({ mode: "readwrite" })) === "granted") { b.textContent = "Agent backups: allowed"; }
        else { h = await showDirectoryPicker({ id: "bbsaves", mode: "readwrite", startIn: "documents" }); await kvSet("dir", h); b.textContent = `Agent backups -> ${h.name}`; }
        const r = await __exportSave({ force: true }); b.textContent += r.file ? " (saved)" : "";
        setTimeout(() => b.remove(), 4000);
      } catch (e) { b.textContent = "Agent: pick backup folder (" + e.name + ")"; }
    };
    document.body.appendChild(b); return "shown";
  };
  window.__exportSave = async ({ everyH = 6, force = false } = {}) => {
    const h = await kvGet("dir"); if (!h) return { need: "folder" };
    const last = (await kvGet("lastExport")) || 0;
    if (!force && Date.now() - last < everyH * 3.6e6) return { skipped: true, name: h.name };
    if ((await h.queryPermission({ mode: "readwrite" })) !== "granted") return { need: "permission", name: h.name };
    const src = await idb("bitburnerSave");
    const data = await reqP(src.transaction("savestring").objectStore("savestring").get("save")); src.close();
    const node = JSON.parse(bb.read("/data/telemetry.txt").trimEnd().split("\n").pop() || "{}").bn || "x";
    const file = `bitburnerSave_${new Date().toISOString().replace(/[:.]/g, "-")}_BN${node}.json.gz`;
    const fh = await h.getFileHandle(file, { create: true }); const w = await fh.createWritable(); await w.write(data); await w.close();
    await kvSet("lastExport", Date.now());
    return { file, bytes: data.length, name: h.name };
  };
  // Fixed-format check-in report (markdown). Everything above "Notes" is generated; the agent only writes Notes.
  const fmt$ = (n) => { const a = Math.abs(n); const s = a >= 1e12 ? (a / 1e12).toFixed(2) + "t" : a >= 1e9 ? (a / 1e9).toFixed(2) + "b" : a >= 1e6 ? (a / 1e6).toFixed(1) + "m" : a >= 1e3 ? (a / 1e3).toFixed(1) + "k" : a.toFixed(0); return (n < 0 ? "-$" : "$") + s; };
  const et = (t) => new Date(t).toLocaleTimeString("en-US", { timeZone: "America/New_York", hour: "numeric", minute: "2-digit" });
  window.__report = (out, st, a, apc, snap) => {
    const tl = (bb.read("/data/telemetry.txt") || "").trimEnd().split("\n").pop(); let t = {}; try { t = JSON.parse(tl); } catch {}
    const installs = (bb.read("/data/install-log.txt") || "").trim().split("\n").filter(Boolean).length;
    const hrs = snap.snap.sinceAug / 3.6e6;
    const status = out.attention.length ? `ATTENTION — ${out.attention.join("; ")}` : out.warn ? `OK (warn: ${out.warn})` : "OK";
    let work = st.work || "idle";
    const m = /^FACTION:(.+)$/.exec(work || "");
    if (m) { const f = m[1], rep = (t.rep || {})[f]; const tgt = (st.next || []).find((n) => n.kind === "rep" && n.faction === f);
      work = `${f} hacking contracts` + (rep != null ? ` — rep ${fmt$(rep).slice(1)}${tgt ? ` / ${fmt$(tgt.target).slice(1)}` : ""}` : ""); }
    const did = [...out.did, ...out.recent].slice(-4).join("; ") || "—";
    const nextAt = et(Date.now() + out.nextMin * 60000);
    return [
      `**Bitburner check-in · ${et(Date.now())} ET** · BN${t.bn || "?"} · install #${installs + 1} (+${hrs.toFixed(1)}h)`,
      "",
      "| | |", "|---|---|",
      `| Status | ${status} |`,
      `| Money | ${fmt$(out.money)}${t.income ? ` (+${fmt$(t.income)}/s)` : ""} |`,
      `| Hack | ${out.hack} · home ${out.homeRam >= 1024 ? out.homeRam / 1024 + "TB" : out.homeRam + "GB"} |`,
      `| Work | ${work} |`,
      `| Augs | ${a.buyable || 0} buyable · install at ${apc.augTrigger || 6}${apc.autoInstall ? " (auto)" : ""} |`,
      `| Activity | ${did} |`,
      `| Next event | ${out.next || "—"} |`,
      `| Predictions | ${(() => { const p = out.pred || {}, l = p.live || {}; return l.n ? `${l.n} scored · ${Math.round(100 * l.inBand)}% in 80% band · ${Math.round(100 * l.within25)}% within ±25%` : `${l.open || 0} open, none scored yet`; })()} |`,
      `| Next check | ${nextAt} ET (+${out.nextMin}m) |`,
    ].join("\n");
  };
  window.__checkin = async (opts = {}) => {
    const out = { attention: [], did: [] };
    // 1. autopilot heartbeat (status file refreshed every ~30s)
    let st = readJ("/data/autopilot-status.txt");
    if (!st || Date.now() - st.t > 120000) {
      const pid = bb.run("agent/autopilot.js");
      out.did.push(pid ? `started autopilot pid ${pid}` : "autopilot start FAILED");
      if (!pid) out.attention.push("autopilot not running and could not start (RAM?)");
    }
    // 2. telemetry heartbeat (last line < 3 min old)
    const tl = (bb.read("/data/telemetry.txt") || "").trimEnd().split("\n").pop();
    let tAge = Infinity; try { tAge = Date.now() - JSON.parse(tl).t; } catch {}
    if (tAge > 180000) { const pid = bb.run("agent/tele-launch.js"); out.did.push(pid ? "relaunched telemetry" : "telemetry relaunch FAILED"); if (!pid) out.attention.push("telemetry down"); }
    // 2b. rolling in-browser save backup (hourly, keeps 24h hourly + 14 daily) in IndexedDB "agentBackups"
    try { const b = await __backupSave(); if (b.saved) out.did.push(`save backup ${Math.round(b.bytes / 1024)}KB (${b.kept} kept)`); }
    catch (e) { out.attention.push("save backup failed: " + e); }
    // 2c. 6-hourly export to the user's folder; if permission lapsed (browser restart), show the re-allow button
    try { const x = await __exportSave(); if (x.file) out.did.push(`exported ${x.file} to ${x.name}`);
      else if (x.need) { __exportButton(); out.warn = `backup folder ${x.need} needed — user must click the green button in the game tab`; } }
    catch (e) { out.warn = "export failed: " + e; }
    // 3. predictions: score, then ETA over autopilot's next actionable targets
    eval(bb.read("agent/predictor.txt"));
    st = readJ("/data/autopilot-status.txt") || st || { flags: [], next: [], augs: {} };
    const specs = (st.next || []).map(({ label, ...s }) => s);
    // predictor v2: resolves past predictions, recalibrates when due, logs a calibrated prediction for every target
    const nx = __pred2.tick(specs); const scored = { length: nx.newlyResolved || 0 };
    if (nx.min == null) nx.min = null;
    const nextLabel = nx && nx.spec ? (st.next.find((n) => n.target === nx.spec.target && n.kind === nx.spec.kind) || {}).label : null;
    // 4. attention rules
    for (const f of st.flags || []) out.attention.push(f);
    const a = st.augs || {};
    const augTrigger = opts.augTrigger ?? 6;
    const apc = readJ("/data/autopilot-config.txt") || {};
    if (apc.autoInstall) {
      // autopilot buys + installs itself; only escalate if it seems stuck
      if (st.installing && Date.now() - st.t > 5 * 60000) out.attention.push("autopilot install started >5 min ago but stack not back");
      if ((a.buyable || 0) >= augTrigger + 4) out.attention.push(`${a.buyable} augs buyable but autopilot has not installed`);
    } else if (a.redPill) out.attention.push("The Red Pill buyable");
    else if ((a.buyable || 0) >= augTrigger) out.attention.push(`${a.buyable} augs buyable (top ${a.top.slice(0, 3).join(", ")})`);
    // 5. snapshot -> next cadence
    const actions = out.did.length + ((st.recent || []).filter((r) => Date.now() - r.t < 20 * 60000).length ? 1 : 0);
    const snap = await bb.snapshot({ actions, etaMin: nx && nx.spec ? nx.min : null, note: out.attention.length ? "attention" : "auto" });
    const rep = __pred2.report();
    Object.assign(out, {
      money: Math.round(st.money), hack: st.hack, homeRam: st.homeRam, work: st.work,
      recent: (st.recent || []).filter((r) => Date.now() - r.t < 60 * 60000).map((r) => r.msg).slice(-6),
      next: nx && nx.spec ? `${nextLabel || JSON.stringify(nx.spec)} in ~${nx.min}m` + (nx.rec ? ` (80%: ${Math.max(0, Math.round((nx.rec.lo - Date.now()) / 6e4))}–${Math.round((nx.rec.hi - Date.now()) / 6e4)}m)` : "") : null,
      augs: a.buyable || 0, pred: rep, newlyScored: scored.length,
      nextMin: apc.autoInstall && !out.attention.length ? Math.max(snap.nextMin, snap.snap.sinceAug < 7.2e6 ? 20 : 45) : snap.nextMin, why: snap.why + (apc.autoInstall ? "; watchdog floor 20/45 (autopilot installs)" : ""),
    });
    // 6. standardized report (agent pastes r.report verbatim, then adds its own Notes line)
    try { out.report = __report(out, st, a, apc, snap); } catch (e) { out.report = "report error: " + e; }
    bb.note(`checkin(auto): $${(out.money / 1e6).toFixed(1)}m hack ${out.hack} | did ${out.did.join("; ") || "-"} | recent ${out.recent.join("; ") || "-"} | attention ${out.attention.join("; ") || "none"} | next ${out.next} | +${out.nextMin}m`);
    return out;
  };
  return "checkin-lib loaded";
})();
